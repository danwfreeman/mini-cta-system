mysql> CREATE FUNCTION fn_is_peak_time_by_day (
    ->   p_time    TIME,
    ->   p_dayType ENUM('weekday','saturday','sunday_holiday')
    -> )
    -> RETURNS BOOLEAN
    -> DETERMINISTIC
    -> BEGIN
    ->   IF p_dayType <> 'weekday' THEN
    ->     RETURN FALSE;
    ->   END IF;
    ->   RETURN (
    ->        (p_time >= '07:00:00' AND p_time < '09:00:00')
    ->     OR (p_time >= '16:00:00' AND p_time < '18:00:00')
    ->   );
