Show quartile grouping for headways by NTILE (show the breakdown in 4 equal parts) PARTITION BY lineID, dayType, direction 
and show the average headway in minutes per group


WITH headways AS (
  SELECT
    l.name AS lineName,
    tt.dayType,
    tt.direction,
    tt.stationID,
    tt.timeOfDay AS arrivalTime,
    TIMESTAMPDIFF(
      MINUTE,
      tt.timeOfDay,
      LEAD(tt.timeOfDay) OVER (
        PARTITION BY tt.lineID, tt.stationID, tt.dayType, tt.direction
        ORDER BY tt.timeOfDay
      )
    ) AS headwayMinutes
  FROM TimeTable tt
  JOIN Line l ON l.lineID = tt.lineID
),
quartiled AS (
  SELECT
    lineName,
    dayType,
    direction,
    headwayMinutes,
    NTILE(4) OVER (
      PARTITION BY lineName, dayType, direction
      ORDER BY headwayMinutes
    ) AS headwayQuartile
  FROM headways
  WHERE headwayMinutes IS NOT NULL
    AND headwayMinutes >= 0
)
SELECT
  lineName,
  dayType,
  direction,
  headwayQuartile,
  AVG(headwayMinutes) AS avgHeadwayMinutes
FROM quartiled
GROUP BY
  lineName, dayType, direction, headwayQuartile
ORDER BY
  lineName, dayType, direction, headwayQuartile;
