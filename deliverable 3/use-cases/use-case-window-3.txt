I want to see a rolling average time between arrivals that go back over the last 15 mins for a line at a station for a specific direction 


mysql> WITH base AS (
       SELECT
         l.name AS lineName,
         s.name AS stationName,
         tt.dayType,
         tt.direction,
         tt.timeOfDay AS arrivalTime,
         TIME_TO_SEC(tt.timeOfDay) AS arrivalSec,
         LEAD(tt.timeOfDay) OVER (
           PARTITION BY tt.lineID, tt.stationID, tt.dayType, tt.direction
           ORDER BY tt.timeOfDay
         ) AS nextArrivalTime,
         TIMESTAMPDIFF(
           MINUTE,
           tt.timeOfDay,
           LEAD(tt.timeOfDay) OVER (
             PARTITION BY tt.lineID, tt.stationID, tt.dayType, tt.direction
             ORDER BY tt.timeOfDay
           )
         ) AS minutesUntilNext
       FROM TimeTable tt
       JOIN Line l    ON l.lineID = tt.lineID
       JOIN Station s ON s.stationID = tt.stationID
       WHERE l.name = 'Blue Line'
         AND tt.dayType = 'weekday'
         AND tt.timeOfDay >= '06:00:00'
         AND tt.timeOfDay <  '13:00:00'
     )
     SELECT
       lineName,
       stationName,
       dayType,
       direction,
       arrivalTime,
       nextArrivalTime,
       minutesUntilNext,
       AVG(minutesUntilNext) OVER (
         PARTITION BY lineName, stationName, dayType, direction
         ORDER BY arrivalSec
         RANGE BETWEEN 900 PRECEDING AND CURRENT ROW
       ) AS rollingAvgHeadway_15min
     FROM base
     WHERE nextArrivalTime IS NOT NULL
     ORDER BY stationName, direction, arrivalTime;
