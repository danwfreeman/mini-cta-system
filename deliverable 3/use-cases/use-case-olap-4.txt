As an operator I want to see the average number of stations per line (train or bus)

mysql> WITH stations_per_line AS (
       SELECT
         l.lineID,
         l.name AS lineName,
         l.mode,           -- 'train' or 'bus'
         COUNT(DISTINCT ls.stationID) AS stationCount
       FROM Line l
       JOIN StationLine ls ON ls.lineID = l.lineID
       GROUP BY l.lineID, l.name, l.mode
     )
     SELECT
       mode,
       AVG(stationCount) AS avgStationsPerLine
     FROM stations_per_line
     GROUP BY mode;
+-------+--------------------+
| mode  | avgStationsPerLine |
+-------+--------------------+
| train |             4.0000 |
| bus   |             2.7500 |
+-------+--------------------+
